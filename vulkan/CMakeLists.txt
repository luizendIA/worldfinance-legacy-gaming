find_package(Vulkan REQUIRED)
find_program(LEGACY_GLSLANG NAMES glslangValidator glslang HINTS "$ENV{VULKAN_SDK}/Bin" REQUIRED)
set(source_header "${PROJECT_SOURCE_DIR}/src/shaders.hpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source_header}")
file(READ "${source_header}" source_text)
string(FIND "${source_text}" "R\"hlsl(" start)
string(FIND "${source_text}" ")hlsl\"" finish)
if(start LESS 0 OR finish LESS start)
  message(FATAL_ERROR "Cannot locate the shared HLSL literal")
endif()
math(EXPR start "${start}+7")
math(EXPR size "${finish}-${start}")
string(SUBSTRING "${source_text}" ${start} ${size} hlsl)
string(REPLACE "cbuffer Camera" "[[vk::push_constant]] cbuffer Camera" hlsl "${hlsl}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/terrain.hlsl" CONTENT "@hlsl@" @ONLY)
function(compile_shader entry profile)
  set(options -D -V --target-env vulkan1.1 --shift-sampler-binding frag 1)
  if(profile STREQUAL "vert")
    list(APPEND options --invert-y)
  endif()
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${entry}.inc"
    COMMAND "${LEGACY_GLSLANG}" ${options} -e ${entry} -S ${profile}
            -o "${CMAKE_CURRENT_BINARY_DIR}/${entry}.inc" --vn "spv_${entry}" "${CMAKE_CURRENT_BINARY_DIR}/terrain.hlsl"
    DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/terrain.hlsl" VERBATIM)
endfunction()
compile_shader(screen vert)
compile_shader(movingTerrain frag)
compile_shader(composite frag)
add_executable(legacy-vulkan main.cpp screen.inc movingTerrain.inc composite.inc)
target_link_libraries(legacy-vulkan PRIVATE Vulkan::Vulkan)
target_include_directories(legacy-vulkan PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${PROJECT_SOURCE_DIR}/src")
target_compile_features(legacy-vulkan PRIVATE cxx_std_17)
target_compile_options(legacy-vulkan PRIVATE /W4 /permissive- /Brepro /utf-8)
target_link_options(legacy-vulkan PRIVATE /Brepro /INCREMENTAL:NO)
