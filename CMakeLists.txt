cmake_minimum_required(VERSION 3.20)
project(legacy_graphics_lab LANGUAGES CXX)
if(NOT MSVC OR NOT WIN32)
  message(FATAL_ERROR "Use an x64 MSVC toolchain with the Windows 10 or 11 SDK.")
endif()
add_executable(legacy-lab src/main.cpp src/render.cpp)
target_compile_features(legacy-lab PRIVATE cxx_std_17)
target_compile_definitions(legacy-lab PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(legacy-lab PRIVATE /W4 /permissive- /Brepro)
target_link_options(legacy-lab PRIVATE /Brepro /INCREMENTAL:NO)
target_link_libraries(legacy-lab PRIVATE d3d11 d3d12 dxgi d3dcompiler)
enable_testing()
add_test(NAME invalid_option COMMAND legacy-lab --invalid)
set_tests_properties(invalid_option PROPERTIES WILL_FAIL TRUE)
add_test(NAME software_smoke COMMAND legacy-lab --warp --smoke)
set_tests_properties(software_smoke PROPERTIES TIMEOUT 90)
